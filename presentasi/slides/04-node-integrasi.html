router.get('/', (req, res) => {
  const page = Math.max(1, parseInt(req.query.page, 10) || 1);
  const perPage = Math.min(50,
    Math.max(1, parseInt(req.query.per_page, 10) || 10));

  const { items, total } = store.list({
    status: req.query.status, q: req.query.q, page, perPage,
  });

  res.json(paginated(req, items,
    { total, page, perPage }, suratResource));
});

router.post('/', apiKey, (req, res) => {
  const errors = validasiSurat(req.body ?? {});
  if (Object.keys(errors).length) {
    throw validationError(errors);       // -> 422
  }

  const surat = store.create(req.body);

  res.status(201)
     .location(`${req.baseUrl}/${surat.id}`)
     .json({ data: suratResource(surat, { detail: true }) });
});

src/validate.jsJS
const POLA_NIM = /^\d{5,15}$/;

export function validasiSurat(body) {
  const errors = {};

  if (!body.nama || String(body.nama).trim() === '') {
    errors.nama = ['Nama wajib diisi.'];
  } else if (String(body.nama).length > 100) {
    errors.nama = ['Nama maksimal 100 karakter.'];
  }

  if (!POLA_NIM.test(String(body.nim ?? ''))) {
    errors.nim = ['NIM harus berupa angka (5-15 digit).'];
  }

  if (!JENIS.includes(body.jenis)) {
    errors.jenis = ['Pilih jenis surat.'];
  }

  if (String(body.keperluan ?? '').trim().length < 10) {
    errors.keperluan = ['Keperluan minimal 10 karakter.'];
  }

  return errors;
}

src/middleware/errorHandler.jsJS
export function errorHandler(err, req, res, next) {
  if (err.type === 'entity.parse.failed') {
    return res.status(400).json({
      message: 'Body request bukan JSON yang valid.',
    });
  }

  if (err instanceof HttpError) {
    return res.status(err.status).json({
      message: err.message,
      ...(err.errors && { errors: err.errors }),
    });
  }

  console.error(err);          // detail hanya di log
  res.status(500).json({
    message: 'Terjadi kesalahan di server.',
  });
}

// errors.js
export class HttpError extends Error {
  constructor(status, message, errors = null) {
    super(message);
    this.status = status; this.errors = errors;
  }
}

app/Services/NotifikasiService.phpPHP
public function statusBerubah(Surat $surat): void
{
    try {
        Http::withHeaders([
                'X-API-Key' => config('services.notifikasi.key'),
            ])
            ->acceptJson()
            ->timeout(5)       // jangan tunggu lama
            ->retry(2, 200)    // coba ulang 2 kali
            ->post(config('services.notifikasi.url'), [
                'event' => 'surat.status_berubah',
                'data'  => ['nomor'  => $surat->nomor,
                            'status' => $surat->status->value],
                'terjadi_pada' => now()->toIso8601String(),
            ])
            ->throw();
    } catch (Throwable $e) {
        // gagal kirim tidak membatalkan perubahan status
        Log::warning('Webhook gagal: '.$e->getMessage());
    }
}

nodejs/integrasi/tarik-dari-laravel.jsJS
const SUMBER = 'http://localhost:8001/api/v1';

async function getJson(url) {
  const res = await fetch(url, {
    headers: { Accept: 'application/json' },
    signal: AbortSignal.timeout(5000),
  });
  const body = await res.json().catch(() => null);
  if (!res.ok) {
    throw new Error(`HTTP ${res.status}: ${body?.message ?? '-'}`);
  }
  return body;
}

async function ambilSemuaSurat() {
  const semua = [];
  let url = `${SUMBER}/surat?per_page=50`;

  while (url) {
    const { data, links } = await getJson(url);
    semua.push(...data);
    url = links.next;      // null = halaman terakhir
  }
  return semua;
}

Keamanan / kunci, batas, origin 30

Empat kontrol yang membedakan demo dari produksi.

01 · AUTENTIKASI

Kunci berbeda untuk peran berbeda

X-API-Key untuk server ke server. Authorization: Bearer untuk aplikasi yang dipakai orang (Sanctum/JWT). Simpan di .env, jangan di Git.

02 · PEMBATASAN

Rate limit 60 permintaan per menit

Laravel: throttle:60,1. Express: express-rate-limit. Tanpa batas, satu skrip bisa menghabiskan kuota server.

03 · TRANSPORT

HTTPS di produksi

API key yang lewat HTTP biasa bisa dibaca di jaringan. Token demo tidak pernah dipakai untuk deployment publik.

04 · CORS

Browser butuh izin, server tidak

Express: app.use(cors()). Laravel: config/cors.php.

CORS hanya berlaku di browser

curl, Http::, dan fetch di Node tidak terkena CORS. Yang diblokir browser adalah JavaScript dari origin lain, bukan permintaan jaringannya.
