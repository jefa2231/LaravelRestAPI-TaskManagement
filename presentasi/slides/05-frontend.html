const BASE = '/api/v1';

async function request(method, path, { query, body } = {}) {
  const url = new URL(BASE + path, location.origin);
  Object.entries(query ?? {}).forEach(([k, v]) => {
    if (v !== '' && v != null) url.searchParams.set(k, v);
  });

  const res = await fetch(url, {
    method,
    headers: {
      Accept: 'application/json',
      ...(body && { 'Content-Type': 'application/json' }),
      ...(method !== 'GET' && { 'X-API-Key': API_KEY }),
    },
    body: body ? JSON.stringify(body) : undefined,
  });

  // fetch TIDAK melempar error untuk 4xx/5xx
  const data = res.status === 204 ? null
    : await res.json().catch(() => null);

  if (!res.ok) throw new ApiError(res.status, data);
  return data;
}
ApiErrorJS
export class ApiError extends Error {
  constructor(status, body) {
    super(body?.message ?? `HTTP ${status}`);
    this.name = 'ApiError';
    this.status = status;
    this.errors = body?.errors ?? {};
  }
}

Form / error 422 per field 34

Kunci errors sama dengan nama field. Itu seluruh triknya.

frontend/app.jsJS
async function kirimForm(form) {
  const tombol = form.querySelector('[type="submit"]');
  tombol.disabled = true;          // cegah klik ganda

  const input = Object.fromEntries(new FormData(form));

  try {
    const { data } = await api.ajukanSurat(input);  // 201
    form.reset();
    await muatDaftar();
    await bukaDetail(data.id);
  } catch (e) {
    if (e instanceof ApiError && e.status === 422) {
      // bersihkan pesan lama, lalu tempel per field
      form.querySelectorAll('[data-error]')
          .forEach((n) => { n.textContent = ''; });

      for (const [field, pesan] of Object.entries(e.errors)) {
        form.querySelector(`[data-error="${field}"]`)
            .textContent = pesan[0];
      }
    } else {
      tampilkanAlert(e.message);   // 401, 500, offline
    }
  } finally {
    tombol.disabled = false;
  }
}

function DaftarSurat() {
  const [list, setList] = useState([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState(null);

  useEffect(() => {
    api.daftarSurat({ per_page: 5 })
      .then(({ data }) => setList(data))
      .catch((e) => setError(e.message))
      .finally(() => setLoading(false));
  }, []);

  if (loading) return <p>Memuat...</p>;
  if (error)   return <p>{error}</p>;

  return list.map((s) => (
    <div key={s.id}>
      {s.nama} — {s.status_label}
    </div>
  ));
}

YANG DIAMATI

Status code, isi errors, dan jumlah permintaan di Network. Tiga hal itu cukup untuk menilai apakah kontraknya konsisten.

Request tanpa Accept: application/json bisa dibalas HTML. Cek tab Network, bukan console.

CORS DI CONSOLE

Checklist / sebelum dikumpulkan 37

Dua belas hal yang membedakan API rapi dari API yang “jalan di laptop saya”.

  • URL kata benda jamak, method sesuai aksi
  • Berversi: /api/v1
  • Status code tepat: 201, 204, 401, 404, 422
  • Selalu JSON, termasuk saat error
  • Sukses: { data }; daftar: { data, links, meta }
  • Error: { message, errors? }
  • Validasi di server, pesan per field
  • Tanggal ISO 8601, nama field konsisten
  • Daftar dipaginasi, per_page dibatasi
  • Autentikasi, HTTPS, dan rate limit
  • Pemanggil API lain: timeout, retry, tangani gagal
  • Kontrak didokumentasikan (KONTRAK-API.md, OpenAPI)
Uji negatif

Kirim body kosong, id yang tidak ada, dan token salah. Kalau ketiganya menghasilkan bentuk error yang sama, kontraknya siap.

Uji ganti backend

Kalau frontend berubah saat backend diganti, ada format yang belum benar-benar sama.

01 · TULIS KONTRAK

Endpoint, field, status code, dan contoh JSON disepakati sebelum kode ditulis.

02 · BANDINGKAN

Respons aktual dibandingkan dengan kontrak, bukan dengan ingatan.

03 · SEBELUM KUMPUL

Jalankan uji negatif dan uji ganti backend, lalu perbarui dokumen.

Latihan / dikerjakan di kelas dan di rumah 38

Enam latihan, dari yang paling mekanis sampai yang paling terbuka.

  1. Uji manual. Dengan curl atau Postman, buat surat baru, proses, lalu selesaikan. Catat status code setiap langkah.
  2. Uji transisi terlarang. Ubah surat berstatus Selesai ke Diproses. Status code apa yang keluar, dan di field mana pesannya?
  3. Endpoint statistik. Tambahkan GET /api/v1/statistik yang mengembalikan jumlah per status. Kedua backend harus menghasilkan JSON identik.
  1. Pakai di UI. Tampilkan angka statistik di tab filter: “Diajukan (1)”.
  2. Simpan notifikasi. Saat Node menerima webhook, simpan ke array dan buat GET /api/v1/notifikasi untuk membacanya.
  3. Tantangan. Ganti API key di frontend dengan login: POST /api/v1/login → token Sanctum, lalu kirim lewat Authorization: Bearer.
Penilaian

Yang dinilai bukan jumlah endpoint, melainkan konsistensi: status code, bentuk error, dan apakah frontend tetap sama saat backend berganti.

YANG DIKUMPULKAN

Repositori Laravel, folder Node.js, dan KONTRAK-API.md yang sudah diperbarui.

BUKTI UJI

Tangkapan layar tab Network untuk 201, 422, dan 401. Satu kurva status code per skenario.

NILAI TAMBAH

Feature test Laravel atau koleksi Postman yang bisa dijalankan ulang oleh penguji.
