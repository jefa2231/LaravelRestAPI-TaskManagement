Status code / hasil yang bisa dibaca mesin 09

Status code memberi tahu hasilnya sebelum body dibaca.

Berhasil
200OK — ada data yang dikembalikan
201Created — POST berhasil, ada Location
202Accepted — diterima, diproses belakangan
204No Content — DELETE berhasil, tanpa body
Gagal
400Bad Request — body bukan JSON valid
401Unauthorized — kunci tidak ada atau salah
403Forbidden — dikenali, tetapi tidak berhak
404Not Found — data atau endpoint tidak ada
422Unprocessable — validasi gagal
500Server Error — bug di server, bukan salah client
Aturan emas

Jangan membalas error dengan 200 + {"ok": false}. fetch, Postman, dan monitoring membaca status code. Kalau semuanya 200, kegagalan tidak terlihat sampai seseorang membuka body.

app/Http/Middleware/ApiKey.phpPHP
final class ApiKey
{
    public function handle(
        Request $request,
        Closure $next,
    ): Response {
        $kunci = (string) $request->header('X-API-Key');
        $asli  = (string) config('services.api_key');

        if ($asli === '' || ! hash_equals($asli, $kunci)) {
            return response()->json(
                ['message' => 'API key tidak valid.'],
                401,
            );
        }

        return $next($request);
    }
}

public function index(IndexSuratRequest $request)
{
    $surat = Surat::query()
        ->when($request->validated('status'),
            fn ($q, $s) => $q->where('status', $s))
        ->when($request->validated('q'),
            fn ($q, $k) => $q->where(fn ($q) => $q
                ->where('nama', 'like', "%{$k}%")
                ->orWhere('nim', 'like', "%{$k}%")))
        ->latest('id')
        ->paginate($request->integer('per_page', 10))
        ->withQueryString();

    return SuratResource::collection($surat);
}

public function store(StoreSuratRequest $request)
{
    $surat = Surat::ajukan($request->validated());

    return (new SuratResource($surat))
        ->response()
        ->setStatusCode(201)
        ->header('Location', route('surat.show', $surat));
}

app/Http/Controllers/Api/SuratController.phpPHP
public function ubahStatus(
    UbahStatusRequest $request,
    Surat $surat,
    NotifikasiService $notifikasi,
) {
    $baru = StatusSurat::from($request->validated('status'));

    if (! $surat->status->bisaBerubahKe($baru)) {
        throw ValidationException::withMessages([
            'status' => "Status {$surat->status->label()} "
                ."tidak bisa diubah ke {$baru->label()}.",
        ]);
    }

    $surat->ubahStatus($baru, (string) $request->validated('catatan'));

    $notifikasi->statusBerubah($surat);

    return new SuratResource($surat);
}

public function destroy(Surat $surat)
{
    $surat->delete();

    return response()->noContent();
}

public function toArray(Request $request): array
{
    return [
        'id'           => $this->id,
        'nomor'        => $this->nomor,
        'nama'         => $this->nama,
        'nim'          => $this->nim,
        'jenis'        => $this->jenis,
        'status'       => $this->status->value,
        'status_label' => $this->status->label(),
        'aksi_berikutnya' => array_map(
            fn (StatusSurat $s) => [
                'status' => $s->value,
                'label'  => $s->labelAksi(),
            ],
            $this->status->berikutnya(),
        ),
        'riwayat' => $this->when(
            ! $request->routeIs('surat.index'),
            fn () => $this->riwayat ?? [],
        ),
        'dibuat_pada' => $this->created_at?->toIso8601String(),
    ];
}

const POLA_NIM = /^\d{5,15}$/;

export function validasiSurat(body) {
  const errors = {};

  if (!body.nama || String(body.nama).trim() === '') {
    errors.nama = ['Nama wajib diisi.'];
  } else if (String(body.nama).length > 100) {
    errors.nama = ['Nama maksimal 100 karakter.'];
  }

  if (!POLA_NIM.test(String(body.nim ?? ''))) {
    errors.nim = ['NIM harus berupa angka (5-15 digit).'];
  }

  if (!JENIS.includes(body.jenis)) {
    errors.jenis = ['Pilih jenis surat.'];
  }

  if (String(body.keperluan ?? '').trim().length < 10) {
    errors.keperluan = ['Keperluan minimal 10 karakter.'];
  }

  return errors;
}

export function errorHandler(err, req, res, next) {
  if (err.type === 'entity.parse.failed') {
    return res.status(400).json({
      message: 'Body request bukan JSON yang valid.',
    });
  }

  if (err instanceof HttpError) {
    return res.status(err.status).json({
      message: err.message,
      ...(err.errors && { errors: err.errors }),
    });
  }

  console.error(err);          // detail hanya di log
  res.status(500).json({
    message: 'Terjadi kesalahan di server.',
  });
}

// errors.js
export class HttpError extends Error {
  constructor(status, message, errors = null) {
    super(message);
    this.status = status; this.errors = errors;
  }
}

async function getJson(url) {
  const res = await fetch(url, {
    headers: { Accept: 'application/json' },
    signal: AbortSignal.timeout(5000),
  });
  const body = await res.json().catch(() => null);
  if (!res.ok) {
    throw new Error(`HTTP ${res.status}: ${body?.message ?? '-'}`);
  }
  return body;
}

  while (url) {

  }
  return semua;
}

const BASE = '/api/v1';

async function request(method, path, { query, body } = {}) {
  const url = new URL(BASE + path, location.origin);
  Object.entries(query ?? {}).forEach(([k, v]) => {
    if (v !== '' && v != null) url.searchParams.set(k, v);
  });

  const res = await fetch(url, {
    method,
    headers: {
      Accept: 'application/json',
      ...(body && { 'Content-Type': 'application/json' }),
      ...(method !== 'GET' && { 'X-API-Key': API_KEY }),
    },
    body: body ? JSON.stringify(body) : undefined,
  });

  // fetch TIDAK melempar error untuk 4xx/5xx
  const data = res.status === 204 ? null
    : await res.json().catch(() => null);

  if (!res.ok) throw new ApiError(res.status, data);
  return data;
}

async function kirimForm(form) {
  const tombol = form.querySelector('[type="submit"]');
  tombol.disabled = true;          // cegah klik ganda

  const input = Object.fromEntries(new FormData(form));

  try {
    const { data } = await api.ajukanSurat(input);  // 201
    form.reset();
    await muatDaftar();
    await bukaDetail(data.id);
  } catch (e) {
    if (e instanceof ApiError && e.status === 422) {
      // bersihkan pesan lama, lalu tempel per field
      form.querySelectorAll('[data-error]')
          .forEach((n) => { n.textContent = ''; });

      for (const [field, pesan] of Object.entries(e.errors)) {
        form.querySelector(`[data-error="${field}"]`)
            .textContent = pesan[0];
      }
    } else {
      tampilkanAlert(e.message);   // 401, 500, offline
    }
  } finally {
    tombol.disabled = false;
  }
}
